{{- /* Source code gets a highlighted panel; terminal output and logs get a distinct near-black block. Fence a block as ```console, ```terminal, ```output or ```log to mark it as output. Optional {title="..."} attribute names either kind. */ -}} {{- $lang := .Type | lower -}} {{- $termLangs := slice "console" "terminal" "shell-session" "output" "log" "logs" -}} {{- if in $termLangs $lang -}} {{- $label := cond (in (slice "log" "logs") $lang) "Log" (cond (eq $lang "output") "Output" "Terminal") -}} {{- with .Attributes.title }}{{ $label = . }}{{ end -}} {{- $body := .Inner | htmlEscape -}} {{- $body = $body | replaceRE `(?m)^(\$|#|>|PS>) ` `$1 ` -}} {{- $body = $body | replaceRE `(?mi)^(.*\b(?:error|fatal|failed|panic|denied|refused)\b.*)$` `$1` -}} {{- $body = $body | replaceRE `(?m)^(.*\b(?:WARN|WARNING)\b.*)$` `$1` -}} {{- else -}} {{- end -}}
