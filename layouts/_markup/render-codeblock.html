{{- /* Source code gets a highlighted panel; terminal output and logs get a distinct near-black block. Fence a block as ```console, ```terminal, ```output or ```log to mark it as output. Optional {title="..."} attribute names either kind. */ -}} {{- $lang := .Type | lower -}} {{- $termLangs := slice "console" "terminal" "shell-session" "output" "log" "logs" -}} {{- if in $termLangs $lang -}} {{- $label := cond (in (slice "log" "logs") $lang) "Log" (cond (eq $lang "output") "Output" "Terminal") -}} {{- with .Attributes.title }}{{ $label = . }}{{ end -}} {{- /* Commands (a prompt line and any lines it continues with a trailing \) are bright, and their prompt can't be selected or copied. Output is dim; warnings and errors are tinted. */ -}} {{- $lines := slice -}} {{- $cont := false -}} {{- $prompt := "" -}} {{- range split (.Inner | htmlEscape) "\n" -}} {{- $ln := . -}} {{- $m := findRESubmatch `^(\$|#|>|PS>) (.*)$` . 1 -}} {{- if $m -}} {{- $ln = printf `%s %s` (index $m 0 1) (index $m 0 2) -}} {{- $prompt = index $m 0 1 -}} {{- $cont = hasSuffix . "\\" -}} {{- else if $cont -}} {{- $ln = printf `%s` . -}} {{- $cont = hasSuffix . "\\" -}} {{- else if findRE `(?i)\b(?:error|fatal|failed|panic|denied|refused)\b` . -}} {{- $ln = printf `%s` . -}} {{- else if findRE `\b(?:WARN|WARNING)\b` . -}} {{- $ln = printf `%s` . -}} {{- end -}} {{- $lines = $lines | append $ln -}} {{- end -}} {{- $body := delimit $lines "\n" -}} {{- /* A block with commands ends at a fresh prompt, the cursor waiting. Not copied. */ -}} {{- with $prompt }}{{ $body = printf "%s\n%s" $body (printf `%s ` .) }}{{ end -}} {{- else -}} {{- end -}}
